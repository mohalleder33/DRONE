const express = require('express');
const router = express.Router();
const auth = require('../middleware/auth');

router.get('/', auth, (req, res) => {
  let platforms = global.mockDB?.platforms || [];
  res.json({ data: platforms, pagination: { page:1, pages:1, total:platforms.length, limit:10 } });
});
router.get('/:id/details', auth, (req, res) => {
  const platform = global.mockDB?.platforms.find(p => p.id === req.params.id);
  if (!platform) return res.status(404).json({ message: 'Platform not found' });
  res.json({ ...platform, equipmentCount:0, ammunitionCount:0, criticalEquipment:[], criticalAmmunition:[], upcomingRotations:[] });
});
